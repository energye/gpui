type InputRouter struct {

	// OnPointer receives every pointer/touch/scroll sample plus the
	// hit-tested target (nil when nothing hit). Handy for gesture arenas.
	OnPointer func(ev input.PointerEvent, target rendering.RenderObject)
	// OnKey receives every logical key event (before focus consumes it).
	OnKey func(ev input.KeyEvent)
	// OnText receives committed text (keyboard chars, paste, IME commits).
	OnText func(ev input.TextEvent)
	// OnIME receives in-progress IME events.
	OnIME func(ev input.IMEEvent)
	// OnEvent receives every normalized event without dedicated handling
	// (window close/move/scale/occluded/hidden/focus/state/theme, frame and
	// sync notices, monitor/locale changes, stylus, pinch/rotate, drag and
	// device events...). It is a fallback observer: events with a dedicated
	// path (pointer/key/text/IME) never reach it, and KindNone never does.
	OnEvent func(ev input.Event)
	// OnDelta receives TextEditingDelta when enableDeltaModel==true.
	OnDelta func(delta textinput.TextEditingDelta)

	// TextEditor is the FALLBACK editable control used when no focused
	// TextEditTarget resolves (single-editor windows / tests). A focused
	// target takes precedence (plan I5).
	TextEditor *textinput.Editor

	// SurroundingUpdates enables periodic set_surrounding_text reporting
	// (context for IME reconversion). OFF by default: each push is a
	// commit_state round-trip and chatty reporting starved the input
	// method (observed: engine switching stopped responding). Enable
	// deliberately when a target needs context-aware IME features.
	SurroundingUpdates bool

	// Has unexported fields.
}
    When attached to a PipelineApp (PipelineOptions.Input), platform events are
    normalized via input.FromPlatform into the cross-platform input.Event and
    routed automatically:

        Pointer/Touch/Scroll → hit test → every control implementing
                                input.PointerHandler on the hit path receives
                                OnPointer; plus the optional OnPointer callback.
        Key → focus manager (when provided) and the optional
                                OnKey callback; logical modifiers tracked.
        Text/IME → optional OnText / OnIME callbacks (textinput
                                milestone consumes these).

    Any RenderObject implementing the input.EventTarget sub-interfaces
    (PointerHandler/KeyHandler/…) is auto-wired — no per-control platform code.
    Attaching an InputRouter is opt-in: windows without it keep the existing
    OnEvent path byte-for-byte, so current examples are unaffected.

func NewInputRouter(hit HitTestFunc, f *focus.FocusManager) *InputRouter
    NewInputRouter creates a router. Pass a hit-test function (typically
    PipelineApp.HitTestPointer) and optionally the focus manager to auto-route

func (r *InputRouter) AttachClipboard(c platform.Clipboard)
    AttachClipboard wires the optional clipboard capability for automatic

func (r *InputRouter) AttachIME(ime platform.IME)
    AttachIME wires the optional IME capability for automatic session
    management: focus transitions of registered TextEditTargets open/close
    sessions; edits refresh the anchor and surrounding text. Re-evaluates the
    current primary immediately; safe before or after SetFocus.

func (r *InputRouter) Clipboard() platform.Clipboard
    Clipboard returns the attached clipboard capability (or nil).

func (r *InputRouter) Modifiers() input.Modifiers
    Modifiers returns the tracked modifier state (for FromPlatform).

func (r *InputRouter) RefreshIMEAnchor()
    RefreshIMEAnchor re-sends the open session's cursor rect and surrounding
    text. Widgets call it after programmatic buffer/caret mutations that did not
    flow through the router (SetText, click-to-place-caret).

func (r *InputRouter) Route(ev input.Event)
    Route dispatches one normalized input event.

func (r *InputRouter) RoutePlatform(ev platform.Event)
    RoutePlatform normalizes a platform event and routes it. The router tracks
    modifier state across events (shift/ctrl/alt/meta press/release).

func (r *InputRouter) SetFocus(f *focus.FocusManager)
    SetFocus attaches the focus manager for key routing. When the IME capability
    is already attached, focus observation starts here.

func (r *InputRouter) SetHitTest(h HitTestFunc)
    SetHitTest overrides the hit-test function (e.g. after the app is bound).

type Options struct {
	// Clear is the PresentClear color (0–1). Default dark gray-blue.
	ClearR, ClearG, ClearB, ClearA float64
	// RenderBackend selects the GPU implementation for this window
	// Zero value means default; GPUI_BACKEND env still wins when set.
	RenderBackend render.Backend
	// ContinuousClear schedules a clear every frame while running (demo only).
	// Production L1 demand mode leaves this false (IDLE until ScheduleFrame).
	ContinuousClear bool
	// OnEvent is called for each host event before default handling.
	OnEvent func(ev platform.Event)
	// MaxFrames stops after N presents when > 0 (tests / timed demos).
	MaxFrames int64
	// RunFor stops the loop after this duration when > 0.
	RunFor time.Duration
}
    Options configures App.

type PipelineApp struct {
	// Has unexported fields.
}
    PipelineApp runs layout/paint → FramePacket → async raster present.
    Steady-state submits never block the UI (SubmitLatest coalesces); open,
    close, snapshot, warm-up and surface teardown synchronize with the raster
    thread by design.

func NewPipelineApp(host platform.Host, root rendering.RenderObject, opts PipelineOptions) *PipelineApp
    NewPipelineApp builds a tree-driven app. Call Open then Run. PhaseSeq
    assigns the pacing phase slot (see scheduler.SetPhaseSeq): windows spawned
    later stagger their frame boundary within the display period so multi-window
    submits spread out instead of colliding. 0 = legacy grid (single-window
    behavior unchanged).

func NewPipelineAppWithPhase(host platform.Host, root rendering.RenderObject, phaseSeq int, opts PipelineOptions) *PipelineApp
    NewPipelineAppWithPhase is NewPipelineApp with an explicit pacing phase slot
    for the window (see NewPipelineApp and scheduler.SetPhaseSeq). Application
    shells pass each window's spawn-order slot here.

func (a *PipelineApp) RetainedSnapshot(path string) bool
    RetainedSnapshot replays the packet of the most recent retained composite on
    the raster thread and writes it as a PNG to path. It reports false when no
    retained frame has been composited yet or the write failed.

    This is the pixel proof for the retained path. SnapshotPath repaints
    the live tree (force, compositeOnly=false), so its PNG describes what a
    FullPaint walk would show — not what the sealed packet actually shipped to
    the screen. Here the same immutable packet is composited again into the
    same context, so a video texture layer lands through scene.TextureLayer →
    render.TextureSource exactly as it did on screen. Evidence-only: replaying
    mid-loop costs one extra composite and re-records cacheable layers into the
    live cache (measured on VR8: frame interval p95 22.5ms without a replay vs
    22.5ms with two).

    THREADING: the replay is a raster job that the UI thread has to submit,
    so a caller that occupies the UI thread also blocks the frame carrying its
    own request — call it from a goroutine, or from the UI thread only when the
    loop is idle-stopped and the request itself drives the submit.

    Run pumps events, ticks animations, builds packets, presents asynchronously.

    ScheduleFrame requests a frame.

    Scheduler returns the frame scheduler (for AddTicker / AnimationController).

    SetDebugRepaint toggles R12b repaint visualization for subsequent presents.

func (a *PipelineApp) SetInputRouter(r *InputRouter)
    SetInputRouter attaches (or replaces) the unified input router after
    construction. The router's hit-test is bound to this app's HitTestPointer.
    Pass nil to detach (fall back to OnEvent input handling).

func (a *PipelineApp) SetOverlay(st *overlay.State)
    SetOverlay attaches an overlay stack (P5d).

func (a *PipelineApp) SetPresentPolicy(policy string)
    SetPresentPolicy sets window present strategy and metrics present_policy.
    Use scheduler.PresentPolicyFullPaint for correctness windows that need a
    full tree repaint every frame.

func (a *PipelineApp) SnapshotAsync(fn func())
    SnapshotAsync schedules fn to run at the end of the next raster frame and
    waits for it. The raster thread owns the render context and swapchain,
    so GPU readback (context.Image / SavePNG) must execute there — calling it
    from the UI thread races the present that owns the texture and aborts wgpu
    ("invalid texture for image copy texture").

func (a *PipelineApp) SubmittedCount() int64
    SubmittedCount returns UI submissions (honesty: submitted vs completed).

func (a *PipelineApp) Target() *render.PresentTarget
    Target returns the present target (nil before Open). RASTER THREAD ONLY:
    render.Context is raster-exclusive and Image() flushes + reads back
    GPU state. UI-thread callers must go through SnapshotAsync (or another
    raster-side hook); sampling the dc off raster races the present it observes.

func (a *PipelineApp) TexturePaintStats() (painted, missed int64)
    TexturePaintStats returns the texture-layer counts of the most recent
    retained composite (painted = sources that drew pixels, missed = ids that
    resolved to nothing). Both are 0 when the last frame carried no texture
    layer. Safe from any thread: published through an atomic by the raster job.

type PipelineOptions struct {
	ClearR, ClearG, ClearB, ClearA float64
	// MaxFrames / RunFor stop conditions.
	MaxFrames int64
	RunFor    time.Duration
	// OnEvent optional.
	OnEvent func(ev platform.Event)
	// When set,
	// pointer/key events are normalized and routed automatically via the
	// router; the per-example OnEvent input handling is not needed. Windows
	// without Input keep the existing OnEvent path unchanged.
	Input *InputRouter
	// IME is the optional input-method capability; when both are set it is
	// attached to Input for automatic session management (plan I4).
	IME platform.IME
	// Clipboard is the optional cross-platform clipboard capability; when
	// both are set it is auto-injected into focused TextEditTargets so
	// apps no longer hand-wire each box (F-B5, X11 nil->fallback).
	Clipboard platform.Clipboard
	// WarmUp runs one full paint before the loop (F11).
	WarmUp bool
	// Overlay is the optional F13 overlay stack (P5d). Hit-test is overlay-first.
	Overlay *overlay.State
	// SaveLayerMaxOps / SaveLayerMaxArea configure the per-frame SaveLayerBudget
	// . 0 = unlimited. Rejections are counted in savelayer_reject.
	SaveLayerMaxOps  int
	SaveLayerMaxArea float64
	// SnapshotPath (optional) saves a GPU readback PNG of the final frame
	// before Close. Use for AA/pixel verification windows: xwd reads the X11
	// backing store, which can lag/diverge from the composited GPU content.
	SnapshotPath string
}
    PipelineOptions configures a demand-driven tree present loop.
